function generateCode() {
    return Math.random()
        .toString(36)
        .substring(2, 8)
        .toUpperCase();
}

function send(ws, data) {
    if (ws.readyState === WebSocket.OPEN) {
        ws.send(JSON.stringify(data));
    }
}

function broadcast(room, data) {
    room.players.forEach(player => {
        send(player.ws, data);
    });
}

function leaderboard(room) {
    return [...room.players.values()]
        .sort((a, b) => b.score - a.score)
        .map((p, index) => ({
            position: index + 1,
            name: p.name,
            score: p.score
        }));
}

function sendLeaderboard(room) {
    broadcast(room, {
        type: "leaderboard",
        leaderboard: leaderboard(room)
    });
}

function startQuestion(room) {

    if (room.questionIndex >= questions.length) {
        room.state = "finished";

        broadcast(room, {
            type: "finished",
            leaderboard: leaderboard(room)
        });

        return;
    }

    room.state = "question";
    room.answers.clear();

    const question = questions[room.questionIndex];

    room.questionStarted = Date.now();

    broadcast(room, {
        type: "question",
        index: room.questionIndex,
        total: questions.length,
        question: question.q,
        answers: question.a,
        time: 20
    });

    clearTimeout(room.timer);

    room.timer = setTimeout(() => {
        endQuestion(room);
    }, 20000);
}

function endQuestion(room) {

    if (room.state !== "question") {
        return;
    }

    clearTimeout(room.timer);

    const question = questions[room.questionIndex];

    room.answers.forEach((answer, playerId) => {

        const player = room.players.get(playerId);

        if (!player) return;

        if (answer.choice === question.correct) {

            const elapsed =
                Date.now() - answer.time;

            const remaining =
                Math.max(
                    0,
                    20000 - elapsed
                );

            const speedBonus =
                Math.round(
                    remaining / 20
                );

            player.score +=
                1000 + speedBonus;

        }

    });

    room.state = "result";

    broadcast(room, {
        type: "result",
        correct: question.correct,
        leaderboard: leaderboard(room)
    });
}

wss.on("connection", ws => {

    ws.on("message", raw => {

        let data;

        try {
            data = JSON.parse(raw);
        } catch {
            return;
        }

        if (data.type === "create") {

            let code = generateCode();

            while (rooms.has(code)) {
                code = generateCode();
            }

            const room = {
                host: ws,
                players: new Map(),
                answers: new Map(),
                questionIndex: 0,
                state: "lobby",
                timer: null,
                questionStarted: null
            };

            rooms.set(code, room);

            ws.roomCode = code;
            ws.isHost = true;

            send(ws, {
                type: "roomCreated",
                code
            });

            sendLeaderboard(room);

            return;
        }

        if (data.type === "join") {

            const room =
                rooms.get(
                    String(data.code || "")
                        .toUpperCase()
                );

            if (!room) {
                send(ws, {
                    type: "error",
                    message: "Lobby nicht gefunden."
                });

                return;
            }

            if (room.state !== "lobby") {
                send(ws, {
                    type: "error",
                    message: "Das Quiz läuft bereits."
                });

                return;
            }

            const id =
                Math.random()
                    .toString(36)
                    .substring(2, 10);

            const player = {
                id,
                name:
                    String(data.name || "Spieler")
                        .substring(0, 20),
                score: 0,
                ws
            };

            room.players.set(id, player);

            ws.roomCode =
                String(data.code)
                    .toUpperCase();

            ws.playerId = id;
            ws.isHost = false;

            send(ws, {
                type: "joined",
                name: player.name
            });

            sendLeaderboard(room);

            return;
        }

        if (data.type === "start") {

            const room =
                rooms.get(ws.roomCode);

            if (!room || !ws.isHost) return;

            startQuestion(room);

            return;
        }

        if (data.type === "answer") {

            const room =
                rooms.get(ws.roomCode);

            if (!room || ws.isHost) return;

            if (room.state !== "question") return;

            if (room.answers.has(ws.playerId)) return;

            room.answers.set(
                ws.playerId,
                {
                    choice: Number(data.choice),
                    time: Date.now()
                }
            );

            send(ws, {
                type: "answerSaved"
            });

            if (
                room.answers.size >=
                room.players.size
            ) {
                endQuestion(room);
            }

            return;
        }

        if (data.type === "next") {

            const room =
                rooms.get(ws.roomCode);

            if (!room || !ws.isHost) return;

            room.questionIndex++;

            startQuestion(room);

            return;
        }

    });

    ws.on("close", () => {

        const room =
            rooms.get(ws.roomCode);

        if (!room) return;

        if (ws.isHost) {

            clearTimeout(room.timer);

            broadcast(room, {
                type: "error",
                message: "Der Host hat das Spiel beendet."
            });

            rooms.delete(ws.roomCode);

            return;
        }

        room.players.delete(ws.playerId);

        sendLeaderboard(room);

    });

});
